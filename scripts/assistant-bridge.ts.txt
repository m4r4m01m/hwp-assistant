
// HWP Assistant downstream bridge. Read-only evidence and reported export only;
// all text mutations still use rhwp's public exact-command SDK.
import { collectTargetEvidence as assistantEvidence } from './document-agent/controller';
import { sha256 as assistantSha256 } from '@noble/hashes/sha2.js';
import { bytesToHex as assistantHex } from '@noble/hashes/utils.js';
const assistantDigest = (value: Uint8Array) => assistantHex(assistantSha256(value));
type AssistantBodyGlyph = [number, number, number]; // logical offset, relative x, advance (px)
type AssistantBodyLine = {start: number; end: number; glyphs: AssistantBodyGlyph[]};
type AssistantBodyParagraph = {
  section: number; paragraph: number; length: number; text_sha256: string; lines: AssistantBodyLine[];
};
let assistantRevision = 0;
const assistantListeners = new Set<() => void>();
eventBus.on('document-mutated', () => {
  assistantRevision += 1;
  for (const listener of assistantListeners) listener();
});

// A read-only rendering receipt for block insertion. It compares the actual
// body glyph positions, independently of HWPX's disposable line-segment cache.
// Page/y positions are deliberately not in the receipt: inserting a block may
// move later text to another page without changing that text's line layout.
function assistantBodyLayoutEvidence() {
  if (!documentAgent) throw new Error('편집기 초기화 중입니다.');
  const revision = assistantRevision;
  const epoch = wasm.documentGeneration;
  const state = documentAgent.getDocumentState();
  const native = wasm.borrowDocumentHandle();
  const page_count = wasm.pageCount;
  if (!native || !Number.isSafeInteger(page_count) || page_count < 1 || page_count > 200)
    throw new Error('문서가 커서 자동 배치 검증을 완료하지 못했습니다. 200쪽 이하 문서에서 다시 시도하세요.');
  const started = performance.now();
  const checkTime = () => {
    // A single native page query cannot be interrupted; stop between queries.
    if (performance.now() - started > 4000)
      throw new Error('문서가 커서 자동 배치 검증을 완료하지 못했습니다. 기존 편집 내용은 그대로입니다.');
  };
  const invalid = () => new Error('기존 본문의 렌더 좌표를 안전하게 확인하지 못했습니다. 기존 편집 내용은 그대로입니다.');
  const sectionCount = native.getSectionCount();
  if (!Number.isSafeInteger(sectionCount) || sectionCount < 1 || sectionCount > 1000) throw invalid();
  const paragraph_counts: number[] = [];
  const paragraphs: AssistantBodyParagraph[] = [];
  const source = new Map<string, {value: AssistantBodyParagraph; chars: string[];
    rows: Array<{page: number; y: number; glyphs: AssistantBodyGlyph[]}>}>();
  let characterCount = 0;
  for (let section = 0; section < sectionCount; section++) {
    const count = native.getParagraphCount(section);
    if (!Number.isSafeInteger(count) || count < 0 || paragraphs.length + count > 10000)
      throw new Error('문서가 커서 자동 배치 검증을 완료하지 못했습니다. 문단 수 제한을 초과했습니다.');
    paragraph_counts.push(count);
    for (let paragraph = 0; paragraph < count; paragraph++) {
      if (paragraph % 100 === 0) checkTime();
      const length = native.getParagraphLength(section, paragraph);
      if (!Number.isSafeInteger(length) || length < 0 || length > 200000) throw invalid();
      const text = native.getTextRange(section, paragraph, 0, length);
      const chars = [...text];
      characterCount += chars.length;
      if (chars.length !== length || characterCount > 200000)
        throw new Error('문서가 커서 자동 배치 검증을 완료하지 못했습니다. 본문 글자 수 제한을 초과했습니다.');
      const value: AssistantBodyParagraph = {section, paragraph, length,
        text_sha256: assistantDigest(new TextEncoder().encode(text)), lines: []};
      paragraphs.push(value);
      source.set(`${section}/${paragraph}`, {value, chars, rows: []});
    }
  }
  let glyphCount = 0;
  for (let page = 0; page < page_count; page++) {
    checkTime();
    const layout = JSON.parse(native.getPageTextLayout(page));
    if (!Array.isArray(layout.runs)) throw invalid();
    for (const run of layout.runs) {
      // Cells and text boxes have a separate coordinate path. Their semantic
      // content/format validation remains the server's existing responsibility.
      if (run.parentParaIdx !== undefined || !Number.isSafeInteger(run.secIdx)
        || !Number.isSafeInteger(run.paraIdx) || !Number.isSafeInteger(run.charStart)
        || typeof run.text !== 'string' || !run.text.length) continue;
      const target = source.get(`${run.secIdx}/${run.paraIdx}`);
      const text = [...run.text];
      if (!target || run.charStart < 0 || run.charStart + text.length > target.value.length
        || text.some((character, index) => target.chars[run.charStart + index] !== character)
        || !Number.isFinite(run.x) || !Number.isFinite(run.y) || !Array.isArray(run.charX)
        || run.charX.length !== text.length + 1 || !run.charX.every(Number.isFinite)) throw invalid();
      glyphCount += text.length;
      if (glyphCount > 200000)
        throw new Error('문서가 커서 자동 배치 검증을 완료하지 못했습니다. 렌더 글자 수 제한을 초과했습니다.');
      let row = target.rows.find(candidate => candidate.page === page && Math.abs(candidate.y - run.y) < 0.2);
      if (!row) { row = {page, y: run.y, glyphs: []}; target.rows.push(row); }
      for (let index = 0; index < text.length; index++) {
        row.glyphs.push([run.charStart + index, run.x + run.charX[index], run.charX[index + 1] - run.charX[index]]);
      }
    }
  }
  for (const {value, rows} of source.values()) {
    value.lines = rows.map(row => {
      row.glyphs.sort((a, b) => a[0] - b[0]);
      const origin = row.glyphs.reduce((left, glyph) => Math.min(left, glyph[1]), Infinity);
      // Native coordinates have 0.1px precision. Keep that precision; callers
      // compare with a small tolerance instead of requiring JSON equality.
      const glyphs = row.glyphs.map(([offset, x, advance]): AssistantBodyGlyph =>
        [offset, Math.round((x - origin) * 10) / 10, Math.round(advance * 10) / 10]);
      return {start: glyphs[0][0], end: glyphs[glyphs.length - 1][0] + 1, glyphs};
    }).sort((a, b) => a.start - b.start || a.end - b.end);
  }
  checkTime();
  const artifact = wasm.exportHwpxWithReport();
  const current = documentAgent.getDocumentState();
  if (artifact.contentLoss.count || revision !== assistantRevision || epoch !== wasm.documentGeneration
    || state.documentEpoch !== current.documentEpoch || state.changeSeq !== current.changeSeq
    || page_count !== wasm.pageCount) throw new Error('본문 배치 조회 중 문서가 바뀌었습니다. 다시 준비하세요.');
  return {revision, epoch, page_count, paragraph_counts, paragraphs, export_sha256: assistantDigest(artifact.bytes)};
}

(window as any).hwpAssistantBridge = {
  async newDocument() { await initPromise; await createNewDocument(); },
  revision: () => assistantRevision,
  documentState() {
    if (!documentAgent) throw new Error('문서가 열려 있지 않습니다.');
    return documentAgent.getDocumentState();
  },
  subscribe(listener: () => void) {
    assistantListeners.add(listener);
    return () => assistantListeners.delete(listener);
  },
  snapshot() {
    if (!documentAgent) throw new Error('편집기 초기화 중');
    const state = documentAgent.getDocumentState();
    const selection = documentAgent.getSelectionContext();
    if (!selection.editable || !selection.target) throw new Error('AI 적용은 단일 서식의 본문 문단만 지원합니다. 표·필드·혼합 서식은 미지원입니다.');
    return { state, selection, evidence: assistantEvidence(wasm, selection.target), revision: assistantRevision };
  },
  exportSnapshot() {
    if (!documentAgent) throw new Error('편집기 초기화 중');
    const artifact = wasm.exportHwpxWithReport();
    return { ...artifact, revision: assistantRevision, epoch: wasm.documentGeneration };
  },
  bodyLayoutEvidence: assistantBodyLayoutEvidence,
  paragraphPage(section: number, paragraph: number) {
    if (!documentAgent || !Number.isSafeInteger(section) || section < 0
      || !Number.isSafeInteger(paragraph) || paragraph < 0 || paragraph >= wasm.getParagraphCount(section))
      throw new Error('렌더링 문단 좌표가 유효하지 않습니다.');
    const position = wasm.getPageOfPosition(section, paragraph);
    if (position.ok !== true || !Number.isSafeInteger(position.page)
      || position.page! < 0 || position.page! >= wasm.pageCount)
      throw new Error('문단의 실제 쪽수를 확인하지 못했습니다.');
    return position.page! + 1;
  },
  // Same iframe and native pagination as the visible editor. This never focuses,
  // edits, or substitutes page 1 when the native lookup fails.
  layoutEvidence(targets: Array<{id: string; section: number; paragraph: number; length: number; text_sha256: string}>) {
    if (!documentAgent || !Array.isArray(targets) || !targets.length || targets.length > 30)
      throw new Error('목차 제목 조회 범위가 유효하지 않습니다.');
    const revision = assistantRevision;
    const epoch = wasm.documentGeneration;
    const state = documentAgent.getDocumentState();
    const artifact = wasm.exportHwpxWithReport();
    if (artifact.contentLoss.count) throw new Error('내용 손실이 있는 문서는 목차를 검증할 수 없습니다.');
    const export_sha256 = assistantDigest(artifact.bytes);
    const page_count = wasm.pageCount;
    const native = wasm.borrowDocumentHandle();
    if (!native || !Number.isSafeInteger(page_count) || page_count < 1 || page_count > 2000)
      throw new Error('렌더된 문서 쪽수가 유효하지 않습니다.');
    const ids = new Set<string>();
    const pages = new Map<number, Array<{secIdx: number; paraIdx: number; charStart: number; text: string}>>();
    const headings = targets.map(target => {
      if (!target || typeof target.id !== 'string' || !target.id || target.id.length > 128 || ids.has(target.id)
        || !Number.isSafeInteger(target.section) || target.section < 0
        || !Number.isSafeInteger(target.paragraph) || target.paragraph < 0
        || !Number.isSafeInteger(target.length) || target.length < 1 || target.length > 300
        || !/^[a-f0-9]{64}$/.test(target.text_sha256))
        throw new Error('목차 제목 좌표 또는 해시가 유효하지 않습니다.');
      ids.add(target.id);
      const length = wasm.getParagraphLength(target.section, target.paragraph);
      if (length !== target.length) throw new Error('렌더된 제목 길이와 고정된 제목 입력이 다릅니다.');
      const text = wasm.getTextRange(target.section, target.paragraph, 0, length);
      const text_sha256 = assistantDigest(new TextEncoder().encode(text));
      if (length !== target.length || [...text].length !== length || text_sha256 !== target.text_sha256)
        throw new Error('렌더된 제목과 고정된 제목 입력이 다릅니다.');
      const position = wasm.getPageOfPosition(target.section, target.paragraph);
      if (position.ok !== true || !Number.isSafeInteger(position.page)
        || position.page! < 0 || position.page! >= page_count)
        throw new Error('제목의 실제 쪽수를 확인하지 못했습니다.');
      const page = position.page!;
      if (!pages.has(page)) {
        const layout = JSON.parse(native.getPageTextLayout(page));
        if (!Array.isArray(layout.runs)) throw new Error('제목의 렌더 좌표가 없습니다.');
        pages.set(page, layout.runs);
      }
      if (!pages.get(page)!.some(run => run.secIdx === target.section && run.paraIdx === target.paragraph
        && run.charStart === 0 && typeof run.text === 'string' && run.text.length > 0
        && text.startsWith(run.text)))
        throw new Error('해당 쪽에서 제목 시작의 렌더 좌표를 확인하지 못했습니다.');
      return {id: target.id, section: target.section, paragraph: target.paragraph, length, text_sha256, page: page + 1};
    });
    const after = wasm.exportHwpxWithReport();
    const current = documentAgent.getDocumentState();
    if (after.contentLoss.count || revision !== assistantRevision || epoch !== wasm.documentGeneration
      || state.documentEpoch !== current.documentEpoch || state.changeSeq !== current.changeSeq
      || page_count !== wasm.pageCount || export_sha256 !== assistantDigest(after.bytes))
      throw new Error('목차 쪽수 조회 중 문서가 바뀌었습니다.');
    return {revision, epoch, page_count, export_sha256, headings};
  },
  warnings: () => wasm.getValidationWarnings(),
  async acknowledge(revision: number, epoch: number, name: string) {
    // Atomic check + markClean occurs before the first await in completeHostSave.
    if (revision !== assistantRevision || epoch !== wasm.documentGeneration) return false;
    await completeHostSave(name);
    return revision === assistantRevision && epoch === wasm.documentGeneration;
  },
};
